{{- /* resolve-image.html - Resolve an image reference to processed asset(s). Looks the path up in the assets pipeline first. For processable raster stills (JPEG/PNG) it emits modern AVIF + WebP derivatives plus a resized original as the universal fallback, all with intrinsic dimensions and content-hashed URLs for long-term caching. Animated GIFs keep their single source (no re-encode). SVGs, externals and anything not found fall back to the original reference untouched, so nothing ever breaks. Input (dict): src – the image reference (required) maxWidth – max width in px before down-scaling raster images (default 1600) Returns (dict): src – fallback URL for the element width – intrinsic width (int) or "" when unknown height – intrinsic height (int) or "" when unknown webp – WebP source URL ("" when not applicable) avif – AVIF source URL ("" when not applicable) external – true for http(s)/protocol-relative/data URIs */ -}} {{- $src := .src -}} {{- $maxWidth := .maxWidth | default 1600 -}} {{- $external := or (hasPrefix $src "http://") (hasPrefix $src "https://") (hasPrefix $src "//") (hasPrefix $src "data:") -}} {{- $out := dict "src" $src "width" "" "height" "" "webp" "" "avif" "" "external" $external -}} {{- if not $external -}} {{- with resources.Get (strings.TrimPrefix "/" $src) -}} {{- if reflect.IsImageResourceProcessable . -}} {{- if in (slice "jpeg" "png") .MediaType.SubType -}} {{- $resize := gt .Width $maxWidth -}} {{- $base := . -}} {{- $pfx := "" -}} {{- if $resize -}} {{- $base = .Resize (printf "%dx q82" $maxWidth) -}} {{- $pfx = printf "resize %dx " $maxWidth -}} {{- end -}} {{- $webp := .Process (printf "%swebp q80" $pfx) -}} {{- $avif := .Process (printf "%savif q55" $pfx) -}} {{- $out = dict "src" $base.RelPermalink "width" $base.Width "height" $base.Height "webp" $webp.RelPermalink "avif" $avif.RelPermalink "external" false -}} {{- else -}} {{- /* Processable but not JPEG/PNG (e.g. animated GIF): keep as-is, just expose dimensions */ -}} {{- $out = dict "src" .RelPermalink "width" .Width "height" .Height "webp" "" "avif" "" "external" false -}} {{- end -}} {{- else -}} {{- /* SVG / ICO / non-processable: publish as-is, dimensions unknown */ -}} {{- $out = dict "src" .RelPermalink "width" "" "height" "" "webp" "" "avif" "" "external" false -}} {{- end -}} {{- end -}} {{- end -}} {{- return $out -}}
